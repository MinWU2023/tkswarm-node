// 执行获取IP
async handleFetchIP() {
if (!this.fetchIPForm.groupId) {
this.$message.warning('请选择分组');
return;
}
if (!this.fetchIPForm.countryId) {
this.$message.warning('请选择国家');

}

this.$message.warning('轮转周期请输入1-120之间的数字');
return;
}

this.fetchingIP = true;
try {
// 根据 countryId 查找对应的 iso 码
const selectedCountry = this.countryList.find(c => c.country_id === this.fetchIPForm.countryId);
if (!selectedCountry) {
this.$message.error('未找到对应的国家信息');
this.fetchingIP = false;
return;
}

const countryIso = selectedCountry.iso; // 获取 iso 码，如 "GB"、"US" 等

const requestData = {
groupId: this.fetchIPForm.groupId,
countryId: countryIso,
areaId: this.fetchIPForm.areaId || '', // 如果为空就传空字符串
rotateCycle: this.fetchIPForm.rotateCycle,
num: this.fetchIPForm.num || 10
};

// 如果城市ID存在且不为空字符串，则添加cityId参数
if (this.fetchIPForm.cityId && this.fetchIPForm.cityId !== '') {
requestData.cityId = this.fetchIPForm.cityId;
}

const response = await apiPost(window.API_BASE_URL + "/tiktok/ip/fetchIP", requestData);

if (response.data.code === 200) {
this.$message.success(response.data.msg || 'IP获取成功');
this.fetchIPDialog = false;
this.refreshData();
this.loadGroup();
} else {
this.$message.error(response.data.msg || '获取失败');
}
} catch (error) {
console.error('获取IP失败:', error);
this.$message.error('请求失败，请稍后重试');
} finally {
this.fetchingIP = false;
}
},

openAddSocksIPDialog() {
if (this.groups.length === 0) {
this.$message.warning('请先创建代理分组');
return;
}
// 重置表单，默认选中当前激活的分组
this.socksIPForm = {
groupId: this.activeGroup !== '0' ? this.activeGroup : '',
ips: ''
};
this.addSocksIPDialog = true;
},

// 处理添加SOCKS IP
async handleAddSocksIP() {
// 验证表单
if (!this.socksIPForm.groupId) {
this.$message.warning('请选择分组');
return;
}
if (!this.socksIPForm.ips.trim()) {
this.$message.warning('请输入SOCKS IP列表');
return;
}

this.socksIPLoading = true;
try {
const requestData = {
groupId: this.socksIPForm.groupId,
ipType: "1",
ipMode: "1",
ips: this.socksIPForm.ips.trim()
};

const response = await apiPost(window.API_BASE_URL + "/tiktok/ip/addn", requestData);

if (response.data.code === 200) {
this.$message.success(response.data.msg || 'SOCKS IP添加成功');
this.addSocksIPDialog = false;
this.socksIPForm.ips = '';
// 刷新数据
this.refreshData();
this.loadGroup();
} else {
this.$message.error(response.data.msg || '添加失败');
}
} catch (error) {
console.error('添加SOCKS IP失败:', error);
this.$message.error('请求失败，请稍后重试');
} finally {
this.socksIPLoading = false;
}
},
