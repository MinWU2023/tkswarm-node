// 更新表单数据
updateFormWithAgentData(data) {
if (!data) return;

// 基础信息
this.form.agentId = data.agent?.agentId || null;
this.form.agentName = data.agent?.agentName || '';
this.form.status = data.agent?.status === 0 ? 0 : 1;
this.form.description = data.agent?.description || '';
this.form.avatar = data.agent?.avatar || this.defaultAvatar;

// 模型配置
if (data.model) {
this.form.modelType = data.model.modelType || 0;
this.form.apiProtocol = data.model.apiProtocol || '';
this.form.modelIdStr = data.model.modelIdStr || '';
this.form.baseUrl = data.model.baseUrl || '';
this.form.apiKey = data.model.apiKey || '';
}

// 对话配置
if (data.conversationConfig) {
this.form.enableRandomOpening = data.conversationConfig.enableRandomOpening === 1;
this.form.roleInfo = data.conversationConfig.roleInfo || '';
this.form.missionObjective = data.conversationConfig.missionObjective || '';
this.form.importantRules = data.conversationConfig.importantRules || '';
this.form.promptLanguage = data.conversationConfig.promptLanguage || 'zh-CN';
this.form.aiReplyLanguage = data.conversationConfig.aiReplyLanguage || 'zh-CN';
this.form.customTone = data.conversationConfig.customTone || '亲切友好';
this.form.temperature = parseFloat(data.conversationConfig.temperature) || 0.7;
this.form.maxTokens = data.conversationConfig.maxTokens || 2000;
}

// 开场白
this.form.openings = data.greetings && data.greetings.length > 0
? data.greetings.map(g => ({
greetingText: g.greetingText || '',
greetingType: g.greetingType || 0
}))
: [{ greetingText: '', greetingType: 0 }];

// 推广资源
if (data.resources && data.resources.length > 0) {
const firstResource = data.resources[0];
this.form.pollEnabled = true;
this.form.resourceType = firstResource.resourceType || 0;
this.form.platform = firstResource.platformId || 1;
this.form.maxLimit = firstResource.maxLimit || 30;
this.form.resources = data.resources.map(r => r.resourceValue).join('\n');
} else {
this.form.pollEnabled = false;
this.form.resources = '';
}

this.form.successTerminateType = data.terminationConfig.successTerminationType || 0;
this.form.failTerminateEnabled = data.terminationConfig.enableFailureTermination === 1;

matchType: k.matchType || 0

this.form.successTerminateKeywords = [{ keyword: '', matchType: 0 }];
}

// 失败终止关键词
this.form.failTerminateKeywords = data.keywords?.filter(k => k.keywordType === 1)?.map(k => ({
keyword: k.keyword || '',
matchType: k.matchType || 0
})) || [{ keyword: '', matchType: 0 }];

this.form.failTerminateKeywords = [{ keyword: '', matchType: 0 }];
}

replyCondition: r.replyCondition || ''

enableFailureTermination: this.form.failTerminateEnabled ? 1 : 0,
failureTerminationType: this.form.failTerminateType
},
aiReplies: [],
keywords: [],
resources: []
};

// 添加AI回复
if (this.form.successTerminateEnabled && this.form.successTerminateType === 1) {
requestData.aiReplies.push({
replyType: 0,
replyContent: this.form.successTerminateReplies[0].replyContent,
replyCondition: this.form.successTerminateReplies[0].replyCondition
});
}

if (this.form.failTerminateEnabled && this.form.failTerminateType === 1) {
requestData.aiReplies.push({
replyType: 1,
replyContent: this.form.failTerminateReplies[0].replyContent,
replyCondition: this.form.failTerminateReplies[0].replyCondition
});
}

// 添加关键词
if (this.form.successTerminateEnabled && this.form.successTerminateType === 0) {
this.form.successTerminateKeywords.forEach(keyword => {
if (keyword.keyword && keyword.keyword.trim()) {
requestData.keywords.push({
keywordType: 0,
keyword: keyword.keyword,
matchType: keyword.matchType
});
}
});
}

if (this.form.failTerminateEnabled && this.form.failTerminateType === 0) {
this.form.failTerminateKeywords.forEach(keyword => {
if (keyword.keyword && keyword.keyword.trim()) {
requestData.keywords.push({
keywordType: 1,
keyword: keyword.keyword,
matchType: keyword.matchType
});
}
});
}

// 添加资源
if (this.form.pollEnabled && this.form.resources) {
const resources = this.form.resources.split('\n')
.filter(r => r.trim())
.map(resource => ({
platformId: parseInt(this.form.platform),
resourceType: parseInt(this.form.resourceType),
resourceValue: resource.trim(),
maxLimit: this.form.maxLimit
}));
requestData.resources = resources;
}

const response = await apiPost('/api/agent/save', requestData);
if (response.data.code === 200) {
this.$message.success('保存成功');
await this.loadAgents();
} else {
this.$message.error(response.data.msg || '保存失败');
}
} catch (error) {
console.error('保存失败:', error);
this.$message.error('网络异常，请检查连接');
} finally {
this.saveLoading = false;
}
});
},

},

}
});
},

// 测试聊天
openTestChatDialog() {
if (!this.currentAgentId) {
this.$message.warning('请先选择一个智能体');
return;
}

this.testChatDialogVisible = true;
this.chatMessages = [];
this.inputMessage = '';
this.isSending = false;

// 自动发送开场白
if (this.form.openings && this.form.openings.length > 0) {
const validOpenings = this.form.openings.filter(o => o.greetingText && o.greetingText.trim());
if (validOpenings.length > 0) {
const opening = this.form.enableRandomOpening
? validOpenings[Math.floor(Math.random() * validOpenings.length)].greetingText
: validOpenings[0].greetingText;

if (opening && opening.trim()) {
this.chatMessages.push({
role: 'assistant',
content: opening.trim(),
timestamp: new Date()
});
}
}
}

this.$nextTick(() => {
this.scrollToBottom();
});
},

closeTestChatDialog() {
this.chatMessages = [];
this.inputMessage = '';
this.isSending = false;
},

async sendMessage() {
if (!this.inputMessage.trim()) {
this.$message.warning('请输入消息');
return;
}

const message = this.inputMessage.trim();
this.inputMessage = '';
this.isSending = true;

// 添加用户消息
this.chatMessages.push({
role: 'user',
content: message,
timestamp: new Date()
});

this.scrollToBottom();

try {
const response = await apiPost('/api/agent/testChat', null, {
params: {
agentId: this.currentAgentId,
message: message
}
});

if (response.data.code === 200) {
this.chatMessages.push({
role: 'assistant',
content: response.data.data.reply || '抱歉，AI暂时无法回复',
timestamp: new Date()
});
} else {
this.chatMessages.push({
role: 'assistant',
content: '抱歉，AI暂时无法回复，请稍后再试',
timestamp: new Date()
});
}
} catch (error) {
console.error('测试聊天失败:', error);
this.chatMessages.push({
role: 'assistant',
content: '抱歉，网络连接异常，请检查网络后重试',
timestamp: new Date()
});
} finally {
this.isSending = false;
this.scrollToBottom();
}
},

formatTime(date) {
const d = new Date(date);
return `${d.getHours().toString().padStart(2, '0')}:${d.getMinutes().toString().padStart(2, '0')}:${d.getSeconds().toString().padStart(2, '0')}`;
},

scrollToBottom() {
this.$nextTick(() => {
const container = this.$refs.chatMessages;
if (container) {
container.scrollTop = container.scrollHeight;
}
});
}
}
