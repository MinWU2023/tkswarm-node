// 处理视频列表数据
processVideoList(videoItems) {
return videoItems.map(item => {
const video = item.video || {};
const stats = item.stats || item.statsV2 || {};

// 尝试获取最佳播放地址
const getBestPlayUrl = () => {
// 优先级：PlayAddrStruct.UrlList > playAddr > downloadAddr
if (video.PlayAddrStruct && video.PlayAddrStruct.UrlList && video.PlayAddrStruct.UrlList.length > 0) {
return video.PlayAddrStruct.UrlList[0];
}
return video.playAddr || video.downloadAddr || '';
};

return {
item_id: item.id || '',
desc: item.desc || '',
create_time: item.createTime || 0,

// 封面图片
cover_url: [
video.originCover ||
video.cover ||
video.dynamicCover ||
''
].filter(url => url),

// 播放地址
play_addr: [getBestPlayUrl()].filter(url => url),

// 统计信息
play_count: parseInt(stats.playCount || stats.diggCount || 0),
like_count: parseInt(stats.diggCount || 0),
comment_count: parseInt(stats.commentCount || 0),
share_count: parseInt(stats.shareCount || 0),

// 视频信息
duration: video.duration || 0,

// 下载信息
download_info: {
download_urls: video.downloadAddr ? [video.downloadAddr] : []
},

// 保留原始数据用于调试
raw_item: item
};
});
},

getVideoPlayUrl(video) {
// 优先级：play_addr > download_info > 原始数据中的地址
if (video.play_addr && video.play_addr.length > 0) {
return video.play_addr[0];
}

if (video.download_info && video.download_info.download_urls && video.download_info.download_urls.length > 0) {
return video.download_info.download_urls[0];
}

// 从原始数据中提取（针对新数据结构）
if (video.raw_item && video.raw_item.video) {
const rawVideo = video.raw_item.video;
return rawVideo.playAddr ||
rawVideo.downloadAddr ||
(rawVideo.PlayAddrStruct && rawVideo.PlayAddrStruct.UrlList && rawVideo.PlayAddrStruct.UrlList[0]);
}

return '';
},

const cookiePairs = [];

for (const [key, value] of Object.entries(cookies)) {
if (key && value !== undefined && value !== null) {

validateAccountParams() {
const params = this.batchAddForm.accountParams;
if (!params.trim()) {
this.accountCount = 0;
this.validationError = '';
return;
}

const lines = params.split('\n').filter(line => line.trim());
this.accountCount = lines.length;

if (lines.length === 0) {
this.validationError = '';
return;
}

// 根据选择的格式进行验证

if (!format) {

return;
}

for (let i = 0; i < lines.length; i++) {
const line = lines[i].trim();
const parts = line.split('----');

let isValid = false;
switch (format) {
case 'format1': // 账号----密码
isValid = parts.length === 2 && parts[0].trim() && parts[1].trim();
break;
case 'format2': // 账号----密码----2fa
isValid = parts.length === 3 && parts[0].trim() && parts[1].trim() && parts[2].trim();
break;
case 'format3': // 账号----密码----CK
isValid = parts.length === 3 && parts[0].trim() && parts[1].trim() && parts[2].trim();
break;
case 'format4': // 账号----CK
isValid = parts.length === 2 && parts[0].trim() && parts[1].trim();
break;
case 'format5': // CK
isValid = parts.length === 1 && parts[0].trim();
break;
case 'format6': // 邮箱----密码----用户名----国家----注册时间----看验证码链接
isValid = parts.length === 6 &&
parts[0].trim() &&
parts[1].trim() &&
parts[2].trim() &&
parts[3].trim() &&
parts[4].trim() &&
parts[5].trim();
break;
case 'format7': // 账号----密码----邮箱验证api
isValid = parts.length === 3 &&
parts[0].trim() &&
parts[1].trim() &&
parts[2].trim();
if (isValid) {
const urlRegex = /^https?:\/\/.+/;
isValid = urlRegex.test(parts[2]) || parts[2].startsWith('http://') || parts[2].startsWith('https://');
if (!isValid) {
this.validationError = `第 ${i + 1} 行邮箱验证API格式不正确，需要以http://或https://开头`;
return;
}
}
break;
case 'format8': // 账号----密码----微软邮箱----微软邮箱密码----clientId----refresh_token
isValid = parts.length === 6 &&
parts[0].trim() &&
parts[1].trim() &&
parts[2].trim() &&
parts[3].trim() &&
parts[4].trim() &&
parts[5].trim();
if (isValid && parts[2]) {
const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
isValid = emailRegex.test(parts[2]);
if (!isValid) {
this.validationError = `第 ${i + 1} 行微软邮箱格式不正确`;
return;
}
}
break;
case 'format9': // IOS全参JSON格式
try {
const jsonData = JSON.parse(line);
if (!jsonData.username && !jsonData.uid && !jsonData.unique_id) {
this.validationError = `第 ${i + 1} 行缺少 username、uid 或 unique_id 字段`;
return;
}
if (!jsonData.cookies && !jsonData.ck) {
this.validationError = `第 ${i + 1} 行缺少 cookies 或 ck 字段`;
return;
}
isValid = true;
} catch (e) {
this.validationError = `第 ${i + 1} 行JSON格式不正确: ${e.message}`;
return;
}
break;
case 'format10': // 安卓全参JSON格式
try {
const jsonData = JSON.parse(line);
// 必须有账号或用户名
if (!jsonData.account && !jsonData.username && !jsonData.user?.username) {
this.validationError = `第 ${i + 1} 行缺少 account、username 或 user.username 字段`;
return;
}
// 必须有cookie
if (!jsonData.cookie && !jsonData.cookies && !jsonData.user?.session_key) {
this.validationError = `第 ${i + 1} 行缺少 cookie 或 cookies 字段`;
return;
}
isValid = true;
} catch (e) {
this.validationError = `第 ${i + 1} 行JSON格式不正确: ${e.message}`;
return;
}
break;
}

if (!isValid) {
this.validationError = `第 ${i + 1} 行格式不正确，请检查`;
return;
}
}

this.validationError = '';
},

parseAccountParams(format, accountParams) {
const lines = accountParams.split('\n').filter(line => line.trim());
const params = [];

lines.forEach(line => {
const parts = line.split('----').map(part => part.trim());
const param = {};

switch (format) {
case 'format1': // 账号----密码
if (parts.length >= 2) {
param.tkUser = parts[0];
param.tkPwd = parts[1];
}
break;
case 'format2': // 账号----密码----2fa
if (parts.length >= 3) {
param.tkUser = parts[0];
param.tkPwd = parts[1];
param.twoFa = parts[2];
}
break;
case 'format3': // 账号----密码----CK
if (parts.length >= 3) {
param.tkUser = parts[0];
param.tkPwd = parts[1];
param.ck = parts[2];
}
break;
case 'format4': // 账号----CK
if (parts.length >= 2) {
param.tkUser = parts[0];
param.ck = parts[1];
}
break;
case 'format5': // CK
if (parts.length >= 1) {
param.ck = parts[0];
param.tkUser = `account_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`;
}
break;
case 'format6': // 邮箱----密码----用户名----国家----注册时间----看验证码链接
if (parts.length >= 6) {
param.tkUser = parts[2]; // 用户名作为账号
param.tkPwd = parts[1]; // 密码
param.country = parts[3]; // 国家
param.accessUrl = parts[5]; // 看验证码链接
// 邮箱信息存储到remark
param.remark = `邮箱:${parts[0]}|注册时间:${parts[4]}`;
}
break;
case 'format7': // 账号----密码----邮箱验证api
if (parts.length >= 3) {
param.tkUser = parts[0];
param.tkPwd = parts[1];
param.accessUrl = parts[2]; // 邮箱验证码API
}
break;
case 'format8': // 账号----密码----微软邮箱----微软邮箱密码----clientId----refresh_token
if (parts.length >= 6) {
param.tkUser = parts[0];
param.tkPwd = parts[1];
// 将微软邮箱相关信息拼接到 accessUrl，格式：邮箱----密码----clientId----refresh_token
param.accessUrl = `${parts[2]}----${parts[3]}----${parts[4]}----${parts[5]}`;
}
break;
case 'format9': // IOS全参（JSON格式）
try {
const jsonData = JSON.parse(line);
param.tkUser = jsonData.username || jsonData.unique_id || `ios_user_${Date.now()}`;
param.tkPwd = jsonData.password || '';

if (jsonData.cookies) {
param.ck = this.convertCookiesToString(jsonData.cookies);
} else if (jsonData.ck) {
if (typeof jsonData.ck === 'string') {
param.ck = jsonData.ck;
} else if (typeof jsonData.ck === 'object') {
param.ck = this.convertCookiesToString(jsonData.ck);
}
}

// 提取设备信息存储到 remark
const deviceInfo = [];
if (jsonData.device_id) deviceInfo.push(`device_id:${jsonData.device_id}`);
if (jsonData.deviceId) deviceInfo.push(`deviceId:${jsonData.deviceId}`);
if (jsonData.odin_id) deviceInfo.push(`odin_id:${jsonData.odin_id}`);
if (jsonData.odinId) deviceInfo.push(`odinId:${jsonData.odinId}`);
if (jsonData.sec_uid) deviceInfo.push(`sec_uid:${jsonData.sec_uid}`);
if (jsonData.secUid) deviceInfo.push(`secUid:${jsonData.secUid}`);
if (jsonData.uid) deviceInfo.push(`uid:${jsonData.uid}`);

if (deviceInfo.length > 0) {
param.remark = deviceInfo.join('|');
}

// 保存其他重要字段到 other_param 或其他字段
if (jsonData.nickname) param.nickname = jsonData.nickname;
if (jsonData.signature) param.signature = jsonData.signature;
if (jsonData.fingerprint) param.fingerprint = jsonData.fingerprint;

} catch (e) {
console.error("解析IOS全参JSON失败:", e);
param.tkUser = `ios_error_${Date.now()}`;
}
break;
case 'format10': // 安卓全参（JSON格式）
try {
const jsonData = JSON.parse(line);

// 解析账号
param.tkUser = jsonData.account || jsonData.username || jsonData.user?.username || `android_user_${Date.now()}`;
param.tkPwd = jsonData.password || '';

// 解析Cookie - 支持多种格式
let cookieStr = '';
if (jsonData.cookie) {
cookieStr = jsonData.cookie;
} else if (jsonData.cookies) {
if (typeof jsonData.cookies === 'string') {
cookieStr = jsonData.cookies;
} else if (typeof jsonData.cookies === 'object') {
cookieStr = this.convertCookiesToString(jsonData.cookies);
}
} else if (jsonData.user?.session_key) {
// 从user对象中提取session_key构建cookie
cookieStr = `sessionid=${jsonData.user.session_key}; sessionid_ss=${jsonData.user.session_key}`;
if (jsonData.user.user_id) {
cookieStr += `; uid_tt=${jsonData.user.user_id}; uid_tt_ss=${jsonData.user.user_id}`;
}
}
param.ck = cookieStr;

// 提取设备信息存储到 remark
const deviceInfo = [];
if (jsonData.device) {
if (jsonData.device.device_id) deviceInfo.push(`device_id:${jsonData.device.device_id}`);
if (jsonData.device.iid) deviceInfo.push(`iid:${jsonData.device.iid}`);
if (jsonData.device.openudid) deviceInfo.push(`openudid:${jsonData.device.openudid}`);

// 提取用户信息
if (jsonData.user) {

if (jsonData.user.sec_user_id) deviceInfo.push(`sec_uid:${jsonData.user.sec_user_id}`);
if (jsonData.user.odin_tt) deviceInfo.push(`odin_tt:${jsonData.user.odin_tt}`);
if (jsonData.user.session_key) deviceInfo.push(`session_key:${jsonData.user.session_key}`);
if (jsonData.user.name) param.nickname = jsonData.user.name;
if (jsonData.user.unique_id) param.friendName = jsonData.user.unique_id;
if (jsonData.user.avatar_url) param.avgImg = jsonData.user.avatar_url;
}

// 提取其他信息
if (jsonData.email) {
deviceInfo.push(`email:${jsonData.email}`);
param.remark = `邮箱:${jsonData.email}`;
}

if (jsonData.country || jsonData.region) {
param.country = jsonData.country || jsonData.region;
}

if (deviceInfo.length > 0) {
param.otherParam = deviceInfo.join('|');
}

// 保存完整的JSON数据到other_param用于后续处理
if (!param.otherParam) {
param.otherParam = JSON.stringify(jsonData);
}

console.log('解析安卓全参成功:', param.tkUser);

} catch (e) {
console.error("解析安卓全参JSON失败:", e);
param.tkUser = `android_error_${Date.now()}`;
param.remark = `解析错误: ${e.message}`;
}
break;


}

if (param.tkUser) {
params.push(param);
}
});

return params;

// 生成二维码

// 开始轮询检查登录状态（5秒一次）

// 清除之前的定时器
this.clearQrCheckInterval();

}, 5000);

// 检查二维码登录状态
checkQrLoginStatus() {
if (!this.qrSessionId) return;

apiGet('/api/tk-account/check-qrcode-status/' + this.qrSessionId)
.then((res) => {
if (res.data.code === 200) {
const data = res.data.data;

if (data.status === 'success') {
// 登录成功
this.qrStatus = 'success';
this.$message.success('扫码登录成功！');

// 清除所有定时器
this.clearAllIntervals();

// 刷新账号列表
this.loadAccounts();

// 3秒后自动关闭对话框
setTimeout(() => {
this.closeQrLoginDialog();
}, 3000);

} else if (data.status === 'scanned') {
// 已扫码
this.qrStatus = 'scanned';
} else if (data.status === 'expired') {
// 二维码过期
this.qrStatus = 'error';
this.$message.warning('二维码已过期');
this.clearAllIntervals();
}
} else {
console.error("检查登录状态失败:", res.data.msg);
}
})
.catch((error) => {
console.error("检查登录状态失败:", error);
});
},

// 开始二维码倒计时
startQrTimer() {
this.qrTimeLeft = 300; // 5分钟
this.qrTimerActive = true;

this.clearQrTimerInterval();

this.qrTimerInterval = setInterval(() => {
this.qrTimeLeft--;

if (this.qrTimeLeft <= 0) {
// 倒计时结束，二维码过期
this.qrStatus = 'error';
this.clearAllIntervals();
this.$message.warning('二维码已过期，请重新获取');
}
}, 1000);
},
