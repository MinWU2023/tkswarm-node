validateHashtagConfig() {
if (!this.hashtagConfig.enabled) return { valid: true };
const content = this.hashtagConfig.content;
if (!content || content.trim() === '') return { valid: false, message: '请填写标签内容' };
const lines = content.split('\n').filter(line => line.trim());
const tagCount = lines.length;
if (tagCount === 0) return { valid: false, message: '请至少填写一个标签' };
const countType = this.hashtagConfig.countType;
if (countType === 'range') {
const minCount = this.hashtagConfig.minCount || 1;
const maxCount = this.hashtagConfig.maxCount || 1;
if (minCount > maxCount) return { valid: false, message: '最小标签数量不能大于最大标签数量' };
if (minCount > tagCount) return { valid: false, message: `最少需要 ${minCount} 个标签，当前只有 ${tagCount} 个` };
if (maxCount > tagCount) return { valid: false, message: `最多只能设置 ${tagCount} 个标签，但您设置了最大 ${maxCount} 个` };
} else if (countType === 'fixed') {
const fixedCount = this.hashtagConfig.fixedCount || 1;
if (fixedCount > tagCount) return { valid: false, message: `每个视频需要 ${fixedCount} 个标签，但当前只有 ${tagCount} 行` };
}
return { valid: true };
