parseISODate: function (dateTimeStr) {
if (!dateTimeStr) return null;
try {
if (typeof dateTimeStr === 'number') {
return new Date(dateTimeStr);
}
if (typeof dateTimeStr === 'string') {
var parts = dateTimeStr.split('T');
if (parts.length !== 2) return null;
var datePart = parts[0];
var timePart = parts[1];
var dateParts = datePart.split('-');
if (dateParts.length !== 3) return null;
var year = parseInt(dateParts[0], 10);
var month = parseInt(dateParts[1], 10) - 1;
var day = parseInt(dateParts[2], 10);
var timeWithZone = timePart.split('+');
var timeMain = timeWithZone[0];
var timeParts = timeMain.split(':');
if (timeParts.length < 2) return null;
var hour = parseInt(timeParts[0], 10);
var minute = parseInt(timeParts[1], 10);
var second = 0;
var millisecond = 0;
if (timeParts.length >= 3) {
var secParts = timeParts[2].split('.');
second = parseInt(secParts[0], 10);
if (secParts.length >= 2) millisecond = parseInt(secParts[1], 10);
}
var date = new Date(Date.UTC(year, month, day, hour, minute, second, millisecond));
if (timeWithZone.length > 1) {
var zoneStr = timeWithZone[1];
if (zoneStr.length >= 4) {
var zoneHour = parseInt(zoneStr.substring(0, 2), 10);
var zoneMinute = parseInt(zoneStr.substring(2, 4), 10);
date.setHours(date.getHours() + zoneHour);
date.setMinutes(date.getMinutes() + zoneMinute);
}
}
return date;
}
return null;
} catch (e) {
console.error('解析日期失败:', e, dateTimeStr);
return null;
}

console.error('解析消息内容失败:', e, message);
originalText = message.content || '';

transformed.roomOwnerSecId = liveShareData.roomOwnerSecId;
transformed.roomOwnerName = liveShareData.roomOwnerName;

.then(function (res) {
if (res.data && res.data.code === 200) {
console.log("标记已读接口调用成功", res.data);

targetAccount = self.accounts[i];

}

if (!targetAccount.friends || typeof targetAccount.friends !== 'object') {
targetAccount.friends = [];
}

var friendsList = targetAccount.friends;
if (!Array.isArray(friendsList)) {
// 如果是对象，尝试转换为数组

friendsList = Object.values(friendsList);
targetAccount.friends = friendsList;
} else {
friendsList = [];
targetAccount.friends = [];
}
}

}

// 清空好友的未读数
targetFriend.unreadCount = 0;
console.log("前端未读数量已清零:", targetFriend);
}
}

// 刷新统计数据
self.loadFriendsCountByStatus(self.selectedAccountId);
} else {
console.error("标记已读接口返回异常:", res.data);
